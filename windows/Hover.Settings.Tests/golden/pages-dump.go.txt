// This is a recipe, not a Go file of the repository (it is saved as .txt so that no Go build or test picks it up).
// It writes windows/Hover.Settings.Tests/golden/pages-go.txt, which PagesAgainstGoTests compares the C# pages with:
//   cp windows/Hover.Settings.Tests/golden/pages-dump.go.txt internal/app/zz_dump_test.go
//   HOVER_RICH_JSON=$PWD/windows/Hover.Settings.Tests/golden/rich-settings.json HOVER_PAGES_OUT=$PWD/windows/Hover.Settings.Tests/golden/pages-go.txt \
//     go test -tags nowayland,nox11,novulkan -run TestZZDumpPages ./internal/app
//   rm internal/app/zz_dump_test.go
// It sets win=true (the Windows words) and no capability (the sandbox, computer use, the agent browser and setup are not on Windows).
func q(s string) string { return fmt.Sprintf("%q", s) }

func dumpControl(c Control) string {
	var p []string
	k := [...]string{"None", "Switch", "Button", "Shortcut", "Segments", "Slider", "Picker", "Text", "Field", "Chips", "Hold"}[c.Kind]
	p = append(p, k)
	if c.ID != "" {
		p = append(p, "id="+c.ID)
	}
	if c.Name != "" {
		p = append(p, "name="+q(c.Name))
	}
	if c.Text != "" {
		p = append(p, "text="+q(c.Text))
	}
	switch c.Kind {
	case CtlSwitch:
		p = append(p, fmt.Sprintf("on=%v", c.On))
	case CtlButton:
		p = append(p, fmt.Sprintf("enabled=%v", c.Enabled))
	case CtlSegments:
		p = append(p, fmt.Sprintf("labels=%q picked=%d", c.Labels, c.Picked))
	case CtlSlider:
		p = append(p, fmt.Sprintf("value=%d min=%d max=%d", c.Value, c.Min, c.Max))
	case CtlPicker:
		var o []string
		for _, x := range c.Options {
			o = append(o, fmt.Sprintf("%q/%v", x.Label, x.On))
		}
		p = append(p, "options=["+strings.Join(o, ", ")+"]")
	case CtlField:
		p = append(p, fmt.Sprintf("placeholder=%q secret=%v on=%v", c.Placeholder, c.Secret, c.On))
	case CtlChips:
		var o []string
		for _, x := range c.Badges {
			o = append(o, fmt.Sprintf("%q/%v", x.Label, x.On))
		}
		var b []string
		for _, x := range c.Buttons {
			b = append(b, fmt.Sprintf("%s:%q:%v", x.ID, x.Text, x.Red))
		}
		p = append(p, "badges=["+strings.Join(o, ", ")+"] buttons=["+strings.Join(b, ", ")+"] open="+c.Open)
	}
	return strings.Join(p, " ")
}

func dumpBlocks(b []Block) string {
	var out strings.Builder
	kinds := [...]string{"Title", "Heading", "Group", "Footnote", "Tiles", "Link", "Lead", "Credits", "Mcp"}
	for _, x := range b {
		switch x.Kind {
		case BlkTitle, BlkLead, BlkFootnote:
			fmt.Fprintf(&out, "[%s] %s\n", kinds[x.Kind], q(x.Text))
		case BlkHeading:
			fmt.Fprintf(&out, "[Heading] %s first=%v\n", q(x.Text), x.First)
		case BlkLink:
			fmt.Fprintf(&out, "[Link] id=%s name=%s icon=%s text=%s dim=%v status=%s\n", x.ID, q(x.Name), x.Icon, q(x.Text), x.Dim, q(x.Status))
		case BlkGroup:
			out.WriteString("[Group]\n")
			for _, r := range x.Rows {
				lead := [...]string{"None", "Tile", "Ring", "Letter", "Mark"}[r.Lead.Kind]
				fmt.Fprintf(&out, "  row %s | sub=%s | enabled=%v | lead=%s/%s/%d/%s | subid=%s | progress=%.2f | ctl: %s\n", q(r.Label), q(r.Sub), r.Enabled, lead, r.Lead.Icon, r.Lead.Tint, r.Lead.Letter, r.SubID, r.Progress, dumpControl(r.Control))
			}
		case BlkTiles:
			for _, t := range x.Tiles {
				fmt.Fprintf(&out, "[Tile] id=%s name=%s from=%s picked=%v\n", t.ID, q(t.Name), q(t.From), t.Picked)
			}
		case BlkCredits:
			out.WriteString("[Credits]\n")
		case BlkMcp:
			out.WriteString("[Mcp]\n")
		}
	}
	return out.String()
}

func TestZZDumpPages(t *testing.T) {
	win, mac = true, false
	dir := t.TempDir()
	var out strings.Builder
	for _, scen := range []string{"default-ready", "default-notready", "voice-local", "rich"} {
		if scen == "rich" {
			b, err := os.ReadFile(os.Getenv("HOVER_RICH_JSON"))
			if err != nil {
				t.Fatal(err)
			}
			os.WriteFile(filepath.Join(dir, scen+".json"), b, 0o644)
		}
		s := core.LoadSettings(filepath.Join(dir, scen+".json"))
		switch scen {
		case "voice-local":
			v := s.Voice()
			v.Enabled, v.Speech = true, core.SpeechLocal
			s.SetVoice(v)
		}
		ready := func(core.AgentTool) *agents.AgentReady { return &agents.AgentReady{Installed: true, SignedIn: true} }
		if scen == "default-notready" {
			ready = func(t core.AgentTool) *agents.AgentReady {
				if t == core.Codex {
					return &agents.AgentReady{Installed: false, Hint: "Codex isn't installed."}
				}
				if t == core.Cursor {
					return &agents.AgentReady{Installed: true, SignedIn: false, Hint: "Sign in to Cursor."}
				}
				return &agents.AgentReady{Installed: true, SignedIn: true}
			}
		}
		in := &Input{Settings: s, Shortcut: s.ScWorkspace().Label(), Reading: noReading, Ready: ready, SystemDark: true,
			VoiceShortcut: s.Voice().Shortcut.Label(), HasSecret: func(string) bool { return false }, SecretsKept: true,
			Live: &Live{Integ: Integ{Caps: Caps{}}}}
		for _, sec := range Sections {
			fmt.Fprintf(&out, "=== %s / %s\n%s", scen, sec.Title(), dumpBlocks(Build(sec, in)))
		}
	}
	os.WriteFile(os.Getenv("HOVER_PAGES_OUT"), []byte(out.String()), 0o644)
}
